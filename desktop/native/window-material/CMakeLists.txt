cmake_minimum_required(VERSION 3.21)
project(omt_window_material LANGUAGES CXX)
add_library(window_material SHARED window-material.cc floating-frame.cc)
target_compile_features(window_material PRIVATE cxx_std_17)
target_compile_definitions(window_material PRIVATE NAPI_VERSION=8 WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_include_directories(window_material PRIVATE "${NODE_HEADERS}")
target_link_libraries(window_material PRIVATE "${NODE_LIBRARY}" user32 gdi32 dwmapi advapi32 comctl32 delayimp)
target_link_options(window_material PRIVATE /DELAYLOAD:node.exe)
set_target_properties(window_material PROPERTIES PREFIX "" SUFFIX ".node" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_options(window_material PRIVATE /W4 /EHsc /utf-8)
