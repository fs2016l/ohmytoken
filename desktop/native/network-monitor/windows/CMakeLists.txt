cmake_minimum_required(VERSION 3.20)
project(ohmytoken_network_monitor LANGUAGES CXX)
add_executable(ohmytoken-network-monitor main.cpp process.cpp pipe.cpp collector.cpp)
target_compile_features(ohmytoken-network-monitor PRIVATE cxx_std_20)
target_compile_definitions(ohmytoken-network-monitor PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_compile_options(ohmytoken-network-monitor PRIVATE /W4 /WX /guard:cf /sdl /utf-8)
target_link_options(ohmytoken-network-monitor PRIVATE /DYNAMICBASE /NXCOMPAT /GUARD:CF)
target_link_libraries(ohmytoken-network-monitor PRIVATE advapi32 tdh ws2_32 shell32)
set_property(TARGET ohmytoken-network-monitor PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
